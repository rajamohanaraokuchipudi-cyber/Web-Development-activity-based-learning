<script>
let count=0;
function addToCart(){
count++;
document.getElementById("cart").textContent=count;
alert("Product added to cart!");
}

function filterProducts(){
let text=document.getElementById("search").value.toLowerCase();
document.querySelectorAll(".product").forEach(function(p){
p.style.display=p.dataset.name.includes(text)?"block":"none";
});
}

function filterCategory(category){
document.getElementById("search").value="";
document.querySelectorAll(".product").forEach(function(p){
p.style.display=(category==="all"||p.dataset.category===category)?"block":"none";
});
}
